#!/usr/bin/env python3 # -*- coding: utf-8 -*- """ VPS 纯数据 API 服务 ==================== 提供实时行情与股票/ETF分红数据，不保存任何持仓信息，无需登录。 接口： GET /api/quotes?codes=601668,513530,... -> JSON 行情 GET /api/dividends?code=601668 -> JSON 分红记录 GET /api/health -> 健康检查 所有响应均包含 CORS 头，允许任意来源访问。 """ import json import re import socket import sys import time import threading from datetime import datetime, timedelta from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer from urllib.parse import urlparse, parse_qs try: import requests except ImportError: sys.exit("缺少依赖 requests, 请先执行: pip install requests") try: from bs4 import BeautifulSoup except ImportError: sys.exit("缺少依赖 beautifulsoup4, 请先执行: pip install beautifulsoup4") if hasattr(sys.stdout, "reconfigure"): sys.stdout.reconfigure(encoding="utf-8", errors="replace") # ========================================================================== # 配置 # ========================================================================== PORT = 8765 # 监听端口 DIVIDEND_CACHE_TTL = 24 * 3600 # 分红缓存时间（秒），默认24小时 HEADERS = { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 " "(KHTML, like Gecko) Chrome/124.0 Safari/537.36", "Referer": "https://quote.eastmoney.com/", "Accept-Language": "zh-CN,zh;q=0.9", } QUOTE_URL = "https://push2.eastmoney.com/api/qt/ulist.np/get" STOCK_DIV_URL = "https://emweb.securities.eastmoney.com/PC_HSF10/BonusFinancing/PageAjax" FUND_DIV_URL = "https://fundf10.eastmoney.com/fhsp_{code}.html" # ========================================================================== # 分红数据缓存（内存 + 线程锁） # ========================================================================== _dividend_cache = {} # code -> (timestamp, records) _dividend_cache_lock = threading.Lock() def fetch_dividends_with_cache(code): """获取分红数据，带24小时缓存，线程安全。""" now = time.time() with _dividend_cache_lock: cached = _dividend_cache.get(code) if cached and (now - cached[0]) < DIVIDEND_CACHE_TTL: return cached[1] # 锁外执行网络请求，避免阻塞其他线程 records = fetch_dividends(code) with _dividend_cache_lock: _dividend_cache[code] = (now, records) return records # -------------------------------------------------------------------------- # 工具函数 # -------------------------------------------------------------------------- def get_lan_ip(): """获取本机局域网 IP，失败则返回 127.0.0.1""" try: s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM) s.connect(("8.8.8.8", 80)) ip = s.getsockname()[0] s.close() return ip except Exception: return "127.0.0.1" def get_with_retry(url, params=None, timeout=10, tries=3): """带重试的 GET 请求，返回 Response 或抛出异常。""" last = None for i in range(tries): try: resp = requests.get(url, params=params, headers=HEADERS, timeout=timeout) resp.raise_for_status() return resp except Exception as e: last = e time.sleep(0.8 * (i + 1)) raise last def classify_code(code): """返回 (类型, 市场)。类型 stock/fund/unknown; 市场 SH/SZ。""" if not re.fullmatch(r"\d{6}", code): return "unknown", None if code[0] in ("6", "5"): return ("stock" if code[0] == "6" else "fund"), "SH" if code[0] in ("0", "3", "1"): return ("stock" if code[0] in ("0", "3") else "fund"), "SZ" return "unknown", None def _d(s): try: return datetime.strptime(str(s).strip()[:10], "%Y-%m-%d").date() except (ValueError, AttributeError): return None # -------------------------------------------------------------------------- # 实时行情 # -------------------------------------------------------------------------- def fetch_quotes_tencent(codes): """腾讯行情接口""" result = {} syms = [] for c in codes: _, mkt = classify_code(c) syms.append(("sh" if mkt == "SH" else "sz") + c) url = "https://qt.gtimg.cn/q=" + ",".join(syms) resp = get_with_retry(url) resp.encoding = "gbk" for m in re.finditer(r'v_(s[hz]\d{6})="([^"]*)"', resp.text): code = m.group(1)[2:] parts = m.group(2).split("~") if len(parts) < 5 or not parts[3]: continue try: price = float(parts[3]) except ValueError: continue result[code] = { "price": price, "name": parts[1], "prev_close": float(parts[4]) if parts[4] else None, "pct": float(parts[32]) if len(parts) > 32 and parts[32] else None, } return result def fetch_quotes_eastmoney(codes): """东财行情接口（备用）""" result = {} secids = [] for c in codes: _, mkt = classify_code(c) secids.append(("1." if mkt == "SH" else "0.") + c) params = {"secids": ",".join(secids), "fields": "f2,f3,f12,f14,f18", "fltt": "2", "invt": "2"} data = get_with_retry(QUOTE_URL, params=params).json() for d in (data.get("data") or {}).get("diff", []): code = d.get("f12") if code is None or d.get("f2") in (None, "-"): continue result[code] = { "price": float(d["f2"]), "name": d.get("f14", ""), "pct": d.get("f3"), "prev_close": d.get("f18"), } return result def fetch_quotes(codes): """获取行情，优先腾讯，失败后回退东财。""" if not codes: return {} try: r = fetch_quotes_tencent(codes) if r: return r except Exception: pass try: return fetch_quotes_eastmoney(codes) except Exception: return {} # -------------------------------------------------------------------------- # 分红数据 # -------------------------------------------------------------------------- def parse_stock_dividends(code): """解析股票（A股）分红数据""" _, mkt = classify_code(code) secu = ("SH" if mkt == "SH" else "SZ") + code j = get_with_retry(STOCK_DIV_URL, params={"code": secu}).json() records = [] for it in j.get("fhyx", []): plan = it.get("IMPL_PLAN_PROFILE", "") or "" ex_date = (it.get("EX_DIVIDEND_DATE") or "").strip() if "派" not in plan or not ex_date: continue m = re.search(r"派\s*([\d.]+)\s*元", plan) if not m: continue records.append({ "ex_date": ex_date[:10], "per_share": round(float(m.group(1)) / 10.0, 5), "year": ex_date[:4], "note": plan, }) records.sort(key=lambda r: r["ex_date"]) return records def parse_fund_dividends(code): """解析场内 ETF/LOF 分红数据""" resp = get_with_retry(FUND_DIV_URL.format(code=code)) resp.encoding = "utf-8" soup = BeautifulSoup(resp.text, "html.parser") records = [] table = soup.find("table", class_="cfxq") if table and table.find("tbody"): for tr in table.find("tbody").find_all("tr"): if "暂无分红信息" in tr.get_text(" ", strip=True): continue tds = tr.find_all("td") if len(tds) < 5: continue ex_date = tds[2].get_text(strip=True) raw = tds[3].get_text(strip=True) m = re.search(r"派现金\s*([\d.]+)", raw) or re.search(r"([\d.]+)\s*元", raw) if not m: continue records.append({ "ex_date": ex_date, "per_share": round(float(m.group(1)) / 10.0, 5), "year": tds[0].get_text(strip=True).replace("年", ""), "note": raw, }) records.sort(key=lambda r: r["ex_date"]) return records def fetch_dividends(code): """根据代码类型获取分红记录""" typ, _ = classify_code(code) if typ == "stock": return parse_stock_dividends(code) if typ == "fund": return parse_fund_dividends(code) return [] # -------------------------------------------------------------------------- # HTTP 请求处理器 # -------------------------------------------------------------------------- class APIHandler(BaseHTTPRequestHandler): def _send_json(self, data, status=200): """发送 JSON 响应，带 CORS 头""" body = json.dumps(data, ensure_ascii=False).encode("utf-8") self.send_response(status) self.send_header("Content-Type", "application/json; charset=utf-8") self.send_header("Access-Control-Allow-Origin", "*") self.send_header("Access-Control-Allow-Headers", "Content-Type") self.send_header("Access-Control-Allow-Methods", "GET, OPTIONS") self.send_header("Content-Length", str(len(body))) self.end_headers() self.wfile.write(body) def _send_error_json(self, message, status=400): self._send_json({"error": message}, status) def do_OPTIONS(self): """处理 CORS 预检请求""" self.send_response(204) self.send_header("Access-Control-Allow-Origin", "*") self.send_header("Access-Control-Allow-Headers", "Content-Type") self.send_header("Access-Control-Allow-Methods", "GET, OPTIONS") self.end_headers() def do_GET(self): parsed = urlparse(self.path) path = parsed.path query = parse_qs(parsed.query) # 健康检查 if path == "/api/health": self._send_json({"status": "ok", "time": datetime.now().isoformat()}) return # 行情接口 if path == "/api/quotes": codes_param = query.get("codes", [""])[0] codes = [c.strip() for c in codes_param.split(",") if c.strip()] # 过滤非法代码格式 valid_codes = [c for c in codes if re.fullmatch(r"\d{6}", c)] if not valid_codes: self._send_error_json("缺少有效的 codes 参数", 400) return quotes = fetch_quotes(valid_codes) self._send_json(quotes) return # 分红接口 if path == "/api/dividends": code = query.get("code", [""])[0].strip() if not re.fullmatch(r"\d{6}", code): self._send_error_json("无效的股票代码", 400) return try: records = fetch_dividends_with_cache(code) self._send_json(records) except Exception as e: self._send_error_json(f"获取分红数据失败: {str(e)}", 500) return # 根路径简单说明 if path == "/": self._send_json({ "service": "VPS 数据 API", "endpoints": [ "/api/health", "/api/quotes?codes=601668,513530", "/api/dividends?code=601668" ] }) return # 其他路径 404 self._send_error_json("Not Found", 404) def log_message(self, format, *args): """自定义日志格式，简洁输出""" sys.stderr.write("%s - - [%s] %s\n" % (self.address_string(), self.log_date_time_string(), format % args)) # -------------------------------------------------------------------------- # 主入口 # -------------------------------------------------------------------------- def main(): print("==============================================") print(" VPS 纯数据 API 服务启动") print("==============================================") lan_ip = get_lan_ip() print(f"监听端口: {PORT}") print(f"本机访问: http://127.0.0.1:{PORT}") print(f"局域网访问: http://{lan_ip}:{PORT}") print("接口列表:") print(" GET /api/health") print(" GET /api/quotes?codes=601668,513530") print(" GET /api/dividends?code=601668") print("按 Ctrl+C 停止服务") server = ThreadingHTTPServer(("0.0.0.0", PORT), APIHandler) try: server.serve_forever() except KeyboardInterrupt: print("\n服务已停止") server.server_close() if __name__ == "__main__": main()
